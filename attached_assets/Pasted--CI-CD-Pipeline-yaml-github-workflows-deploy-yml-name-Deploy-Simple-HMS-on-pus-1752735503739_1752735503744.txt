### Database Backup Strategy

```bash
#!/bin/bash
# backup.sh - Automated database backup script

set -e

# Configuration
DB_HOST="localhost"
DB_PORT="5432"
DB_NAME="simple_hms"
DB_USER="hms_user"
BACKUP_DIR="/opt/backups/simple-hms"
RETENTION_DAYS=30
DATE=$(date +%Y%m%d_%H%M%S)

# Create backup directory if it doesn't exist
mkdir -p "$BACKUP_DIR"

# Full database backup
echo "Starting database backup..."
pg_dump -h "$DB_HOST" -p "$DB_PORT" -U "$DB_USER" -d "$DB_NAME" \
    --verbose --clean --no-owner --no-privileges \
    | gzip > "$BACKUP_DIR/full_backup_$DATE.sql.gz"

# Schema-only backup for each hospital
echo "Creating schema backups..."
for schema in $(psql -h "$DB_HOST" -p "$DB_PORT" -U "$DB_USER" -d "$DB_NAME" \
    -t -c "SELECT schema_name FROM information_schema.schemata 
           WHERE schema_name LIKE 'hospital_%'"); do
    
    schema=$(echo $schema | xargs)  # Trim whitespace
    pg_dump -h "$DB_HOST" -p "$DB_PORT" -U "$DB_USER" -d "$DB_NAME" \
        --schema="$schema" --verbose --clean --no-owner --no-privileges \
        | gzip > "$BACKUP_DIR/schema_${schema}_$DATE.sql.gz"
done

# Remove old backups
echo "Cleaning up old backups..."
find "$BACKUP_DIR" -name "*.sql.gz" -mtime +$RETENTION_DAYS -delete

# Upload to cloud storage (optional)
if [ "$CLOUD_BACKUP_ENABLED" = "true" ]; then
    echo "Uploading to cloud storage..."
    aws s3 cp "$BACKUP_DIR/full_backup_$DATE.sql.gz" \
        "s3://your-backup-bucket/simple-hms/$(date +%Y/%m/)/"
fi

echo "Backup completed successfully!"

# Send notification
curl -X POST "$SLACK_WEBHOOK_URL" \
    -H 'Content-type: application/json' \
    --data "{\"text\":\"Simple HMS backup completed successfully - $DATE\"}"
```

```bash
#!/bin/bash
# health_check.sh - System health monitoring

# Configuration
API_URL="http://localhost:8080/api"
DB_HOST="localhost"
DB_PORT="5432"
DB_NAME="simple_hms"
SLACK_WEBHOOK=""

# Function to send alerts
send_alert() {
    local message="$1"
    echo "ALERT: $message"
    
    if [ -n "$SLACK_WEBHOOK" ]; then
        curl -X POST "$SLACK_WEBHOOK" \
            -H 'Content-type: application/json' \
            --data "{\"text\":\"🚨 Simple HMS Alert: $message\"}"
    fi
}

# Check API health
echo "Checking API health..."
if ! curl -sf "$API_URL/actuator/health" > /dev/null; then
    send_alert "API health check failed - service may be down"
    exit 1
fi

# Check database connectivity
echo "Checking database connectivity..."
if ! pg_isready -h "$DB_HOST" -p "$DB_PORT" -d "$DB_NAME" > /dev/null; then
    send_alert "Database connectivity check failed"
    exit 1
fi

# Check disk space
echo "Checking disk space..."
DISK_USAGE=$(df / | awk 'NR==2 {print $5}' | sed 's/%//')
if [ "$DISK_USAGE" -gt 85 ]; then
    send_alert "Disk usage is above 85% ($DISK_USAGE%)"
fi

# Check memory usage
echo "Checking memory usage..."
MEMORY_USAGE=$(free | awk 'NR==2{printf "%.0f", $3*100/$2}')
if [ "$MEMORY_USAGE" -gt 90 ]; then
    send_alert "Memory usage is above 90% ($MEMORY_USAGE%)"
fi

# Check application logs for errors
echo "Checking application logs..."
ERROR_COUNT=$(tail -1000 /opt/simple-hms/logs/simple-hms.log | grep -c "ERROR" || true)
if [ "$ERROR_COUNT" -gt 10 ]; then
    send_alert "High error count detected in logs: $ERROR_COUNT errors in last 1000 lines"
fi

echo "Health check completed successfully!"
```

```sql
-- Migration Script Template
-- V{VERSION}__Description.sql

-- Pre-migration checks
DO $
BEGIN
    -- Check if migration has already been applied
    IF EXISTS (SELECT 1 FROM schema_version WHERE version = '{VERSION}') THEN
        RAISE EXCEPTION 'Migration version {VERSION} has already been applied';
    END IF;
    
    -- Check prerequisites
    IF NOT EXISTS (SELECT 1 FROM schema_version WHERE version = '{PREVIOUS_VERSION}') THEN
        RAISE EXCEPTION 'Previous migration version {PREVIOUS_VERSION} not found';
    END IF;
END $;

-- Start transaction
BEGIN;

-- Create backup table (if modifying existing data)
CREATE TABLE patients_backup_v{VERSION} AS SELECT * FROM patients;

-- Apply changes
ALTER TABLE patients ADD COLUMN IF NOT EXISTS middle_name VARCHAR(100);
ALTER TABLE patients ADD COLUMN IF NOT EXISTS preferred_name VARCHAR(100);

-- Update existing data (if needed)
UPDATE patients SET preferred_name = first_name WHERE preferred_name IS NULL;

-- Validate changes
DO $
DECLARE
    record_count INTEGER;
BEGIN
    SELECT COUNT(*) INTO record_count FROM patients WHERE preferred_name IS NULL;
    IF record_count > 0 THEN
        RAISE EXCEPTION 'Migration validation failed: % records have null preferred_name', record_count;
    END IF;
END $;

-- Record migration
INSERT INTO schema_version (version, applied_at, description) 
VALUES ('{VERSION}', NOW(), 'Add middle_name and preferred_name to patients');

-- Clean up backup table (optional, keep for rollback)

COMMIT;
```

```sql
-- Materialized Views for Reports
CREATE MATERIALIZED VIEW monthly_revenue_summary AS
SELECT 
    DATE_TRUNC('month', bill_date) as month,
    COUNT(*) as total_bills,
    SUM(total_amount) as total_revenue,
    SUM(paid_amount) as total_paid,
    AVG(total_amount) as avg_bill_amount
FROM bills 
WHERE payment_status != 'CANCELLED'
GROUP BY DATE_TRUNC('month', bill_date)
ORDER BY month DESC;

-- Refresh materialized view daily
CREATE OR REPLACE FUNCTION refresh_monthly_revenue_summary()
RETURNS void AS $
BEGIN
    REFRESH MATERIALIZED VIEW CONCURRENTLY monthly_revenue_summary;
END;
$ LANGUAGE plpgsql;
```

```javascript
// Secure Local Storage
class SecureStorage {
  static encrypt(data) {
    return btoa(JSON.stringify(data));
  }
  
  static decrypt(encryptedData) {
    try {
      return JSON.parse(atob(encryptedData));
    } catch {
      return null;
    }
  }
  
  static setItem(key, value) {
    const encrypted = this.encrypt(value);
    localStorage.setItem(key, encrypted);
  }
  
  static getItem(key) {
    const encrypted = localStorage.getItem(key);
    return encrypted ? this.decrypt(encrypted) : null;
  }
}
```

```java
// Custom Logging Aspect
@Aspect
@Component
@Slf4j
public class LoggingAspect {

    @Around("@annotation(Loggable)")
    public Object logExecutionTime(ProceedingJoinPoint joinPoint) throws Throwable {
        long start = System.currentTimeMillis();
        
        String className = joinPoint.getSignature().getDeclaringTypeName();
        String methodName = joinPoint.getSignature().getName();
        Object[] args = joinPoint.getArgs();
        
        try {
            log.info("Executing {}.{}() with args: {}", className, methodName, args);
            Object result = joinPoint.proceed();
            
            long executionTime = System.currentTimeMillis() - start;
            log.info("Completed {}.{}() in {} ms", className, methodName, executionTime);
            
            return result;
        } catch (Exception e) {
            long executionTime = System.currentTimeMillis() - start;
            log.error("Exception in {}.{}() after {} ms: {}", 
                className, methodName, executionTime, e.getMessage(), e);
            throw e;
        }
    }
    
    @Before("@annotation(AuditLog)")
    public void auditLog(JoinPoint joinPoint) {
        String action = joinPoint.getSignature().getName();
        String entityType = joinPoint.getSignature().getDeclaringTypeName();
        
        // Get current user and hospital context
        String userId = SecurityContextHolder.getContext().getAuthentication().getName();
        String hospitalId = TenantContext.getCurrentHospitalId();
        
        auditService.logAction(userId, hospitalId, action, entityType);
    }
}

public class RequestLoggingFilter implements Filter {
    
    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {
        
        HttpServletRequest httpRequest = (HttpServletRequest) request;

        String correlationId = UUID.randomUUID().toString();

        MDC.put("method", httpRequest.getMethod());

    }
```

```java
    public ResponseEntity<AuthResponse> login(@RequestBody LoginRequest request) {
```

```javascript
// contexts/AuthContext.js
import React, { createContext, useContext, useReducer, useEffect } from 'react';
import { authService } from '../services/auth.service';

const AuthContext = createContext();

const authReducer = (state, action) => {
  switch (action.type) {
    case 'LOGIN_SUCCESS':
      return {
        ...state,
        isAuthenticated: true,
        user: action.payload.user,
        hospitalAccess: action.payload.hospitalAccess,
        loading: false
      };
    case 'LOGOUT':
      return {
        ...state,
        isAuthenticated: false,
        user: null,
        hospitalAccess: [],
        loading: false
      };
    case 'SET_LOADING':
      return { ...state, loading: action.payload };
    default:
      return state;
  }
};

export const AuthProvider = ({ children }) => {
  const [state, dispatch] = useReducer(authReducer, {
    isAuthenticated: false,
    user: null,
    hospitalAccess: [],
    loading: true
  });

  useEffect(() => {
    const token = localStorage.getItem('token');
    if (token) {
      authService.validateToken(token)
        .then(response => {
          dispatch({
            type: 'LOGIN_SUCCESS',
            payload: response.data
          });
        })
        .catch(() => {
          localStorage.removeItem('token');
          dispatch({ type: 'SET_LOADING', payload: false });
        });
    } else {
      dispatch({ type: 'SET_LOADING', payload: false });
    }
  }, []);

  const login = async (credentials) => {
    try {
      const response = await authService.login(credentials);
      localStorage.setItem('token', response.data.token);
      dispatch({
        type: 'LOGIN_SUCCESS',
        payload: response.data
      });
      return response.data;
    } catch (error) {
      throw error;
    }
  };

  const logout = () => {
    localStorage.removeItem('token');
    localStorage.removeItem('currentHospitalId');
    dispatch({ type: 'LOGOUT' });
  };

  return (
    <AuthContext.Provider value={{ ...state, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
};

// contexts/HospitalContext.js
import React, { createContext, useContext, useState, useEffect } from 'react';
import { hospitalService } from '../services/hospital.service';
import { useAuth } from './AuthContext';

const HospitalContext = createContext();

export const HospitalProvider = ({ children }) => {
  const [currentHospital, setCurrentHospital] = useState(null);
  const [hospitals, setHospitals] = useState([]);
  const { user, hospitalAccess } = useAuth();

  useEffect(() => {
    if (hospitalAccess?.length > 0) {
      setHospitals(hospitalAccess.map(access => access.hospital));
      
      // Set default hospital
      const savedHospitalId = localStorage.getItem('currentHospitalId');
      if (savedHospitalId) {
        const hospital = hospitalAccess.find(access => 
          access.hospital.id === parseInt(savedHospitalId)
        )?.hospital;
        if (hospital) {
          setCurrentHospital(hospital);
          return;
        }
      }
      
      // Set first hospital as default
      const defaultAccess = hospitalAccess.find(access => access.isDefaultHospital) 
        || hospitalAccess[0];
      setCurrentHospital(defaultAccess.hospital);
    }
  }, [hospitalAccess]);

  const switchHospital = async (hospitalId) => {
    try {
      await hospitalService.switchHospital(hospitalId);
      const hospital = hospitals.find(h => h.id === hospitalId);
      setCurrentHospital(hospital);
      localStorage.setItem('currentHospitalId', hospitalId.toString());
    } catch (error) {
      console.error('Failed to switch hospital:', error);
    }
  };

  return (
    <HospitalContext.Provider value={{
      currentHospital,
      hospitals,
      switchHospital
    }}>
      {children}
    </HospitalContext.Provider>
  );
};

export const useHospital = () => {
  const context = useContext(HospitalContext);
  if (!context) {
    throw new Error('useHospital must be used within a HospitalProvider');
  }
  return context;
};
```

## 🏥 Project Overview

Simple HMS is a comprehensive multi-tenant hospital management system designed to streamline healthcare operations across multiple facilities within a single organization. The system supports role-based access control and enables healthcare professionals to work seamlessly across different hospitals with unified login credentials.

### Key Capabilities
- **Multi-Hospital Management**: Single organization managing multiple clinics with schema-based data separation
- **Cross-Facility Practice**: Healthcare professionals can work at multiple hospitals with unified access
- **Role-Based Security**: Granular permissions for Admin, Doctor, Receptionist, and Medicine Clerk
- **Complete Patient Lifecycle**: From registration through treatment to follow-up care
- **Integrated Operations**: Appointments, billing, inventory, and reporting in one system

    payment_status VARCHAR(20) DEFAULT 'UNPAID', -- Paid, Unpaid, Partial, Overdue
    payment_method VARCHAR(50),
    financial_category VARCHAR(50), -- Self_Pay, Insured, Government
    insurance_claim_status VARCHAR(50),
    insurance_claim_number VARCHAR(100),
    insurance_approved_amount DECIMAL(10,2),
    generated_by INTEGER,
    remarks TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Bill items
CREATE TABLE bill_items (
    bill_item_id SERIAL PRIMARY KEY,
    bill_id INTEGER REFERENCES bills(bill_id),
    item_type VARCHAR(50), -- Consultation, Lab_Test, Medicine, Procedure
    item_code VARCHAR(50),
    item_description TEXT,
    quantity INTEGER DEFAULT 1,
    unit_price DECIMAL(10,2),
    total_price DECIMAL(10,2),
    discount_percentage DECIMAL(5,2) DEFAULT 0,
    tax_percentage DECIMAL(5,2) DEFAULT 0
);

-- Lab test orders
CREATE TABLE lab_test_orders (
    lab_order_id SERIAL PRIMARY KEY,
    order_number VARCHAR(50) UNIQUE,
    patient_id INTEGER REFERENCES patients(patient_id),
    doctor_id INTEGER REFERENCES doctors(doctor_id),
    visit_id INTEGER REFERENCES visits(visit_id),
    ordered_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    sample_collection_date TIMESTAMP,
    reported_date TIMESTAMP,
    priority VARCHAR(20) DEFAULT 'ROUTINE', -- Routine, Urgent, STAT
    status VARCHAR(20) DEFAULT 'ORDERED', -- Ordered, Sample_Collected, In_Progress, Completed
    lab_technician_id INTEGER,
    total_cost DECIMAL(10,2),
    notes TEXT
);

-- Lab test items
CREATE TABLE lab_test_items (
    lab_test_item_id SERIAL PRIMARY KEY,
    lab_order_id INTEGER REFERENCES lab_test_orders(lab_order_id),
    test_id INTEGER REFERENCES test_master(test_id),
    test_name VARCHAR(255),
    result VARCHAR(500),
    normal_range VARCHAR(100),
    unit VARCHAR(50),
    abnormal_flag BOOLEAN DEFAULT FALSE,
    comments TEXT,
    verified_by INTEGER,
    verified_at TIMESTAMP
);

-- Test master (reference data)
CREATE TABLE test_master (
    test_id SERIAL PRIMARY KEY,
    test_name VARCHAR(255) NOT NULL,
    test_code VARCHAR(50) UNIQUE,
    category VARCHAR(100),
    normal_range VARCHAR(100),
    unit VARCHAR(50),
    cost DECIMAL(10,2),
    sample_type VARCHAR(50), -- Blood, Urine, Stool, etc.
    preparation_instructions TEXT,
    reporting_time_hours INTEGER,
    is_active BOOLEAN DEFAULT TRUE
);

-- Medicines
CREATE TABLE medicines (
    medicine_id SERIAL PRIMARY KEY,
    medicine_name VARCHAR(255) NOT NULL,
    generic_name VARCHAR(255),
    brand_name VARCHAR(255),
    dosage_form VARCHAR(100), -- Tablet, Syrup, Injection, etc.
    strength VARCHAR(100),
    manufacturer VARCHAR(255),
    category VARCHAR(100),
    unit_price DECIMAL(10,2),
    selling_price DECIMAL(10,2),
    reorder_level INTEGER,
    description TEXT,
    common_side_effects TEXT,
    drug_interactions TEXT,
    contraindications TEXT,
    storage_conditions TEXT,
    prescription_required BOOLEAN DEFAULT TRUE,
    pregnancy_category VARCHAR(5),
    drug_classification VARCHAR(50), -- Narcotic, Controlled, OTC
    is_active BOOLEAN DEFAULT TRUE
);

-- Medicine stock
CREATE TABLE medicine_stock (
    stock_id SERIAL PRIMARY KEY,
    medicine_id INTEGER REFERENCES medicines(medicine_id),
    batch_number VARCHAR(100),
    expiry_date DATE,
    quantity_available INTEGER,
    quantity_received INTEGER,
    received_date DATE,
    supplier_name VARCHAR(255),
    purchase_cost DECIMAL(10,2),
    mrp DECIMAL(10,2),
    manufacturing_date DATE,
    status VARCHAR(20) DEFAULT 'AVAILABLE' -- Available, Expired, Recalled
);

-- Medicine dispensing
CREATE TABLE medicine_issue (
    issue_id SERIAL PRIMARY KEY,
    issue_number VARCHAR(50) UNIQUE,
    patient_id INTEGER REFERENCES patients(patient_id),
    visit_id INTEGER REFERENCES visits(visit_id),
    medicine_id INTEGER REFERENCES medicines(medicine_id),
    batch_number VARCHAR(100),
    quantity_issued INTEGER,
    issue_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    prescribed_by INTEGER,
    dispensed_by INTEGER,
    dosage_instructions TEXT,
    duration_days INTEGER,
    refills_allowed INTEGER DEFAULT 0,
    refills_issued INTEGER DEFAULT 0,
    prescription_status VARCHAR(20) DEFAULT 'NEW', -- New, Filled, Refilled, Expired
    unit_price DECIMAL(10,2),
    total_amount DECIMAL(10,2)
);

-- Documents
CREATE TABLE documents (
    document_id SERIAL PRIMARY KEY,
    patient_id INTEGER REFERENCES patients(patient_id),
    visit_id INTEGER REFERENCES visits(visit_id),
    document_type VARCHAR(100), -- Prescription, Report, Consent, Insurance_Card
    document_name VARCHAR(255),
    file_name VARCHAR(255),
    file_path VARCHAR(500),
    file_size BIGINT,
    mime_type VARCHAR(100),
    upload_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    uploaded_by INTEGER,
    expiry_date DATE,
    document_status VARCHAR(20) DEFAULT 'ACTIVE',
    is_verified BOOLEAN DEFAULT FALSE,
    remarks TEXT
);

-- Follow-ups
CREATE TABLE follow_ups (
    follow_up_id SERIAL PRIMARY KEY,
    patient_id INTEGER REFERENCES patients(patient_id),
    visit_id INTEGER REFERENCES visits(visit_id),
    doctor_id INTEGER REFERENCES doctors(doctor_id),
    suggested_date DATE,
    actual_date DATE,
    reason TEXT,
    status VARCHAR(20) DEFAULT 'PENDING', -- Pending, Completed, Missed, Cancelled
    remarks TEXT,
    created_by INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Clinical notes
CREATE TABLE notes (
    note_id SERIAL PRIMARY KEY,
    patient_id INTEGER REFERENCES patients(patient_id),
    visit_id INTEGER REFERENCES visits(visit_id),
    note_type VARCHAR(50), -- Doctor_Note, Nurse_Note, Progress_Note
    title VARCHAR(255),
    content TEXT,
    is_visible_to_patient BOOLEAN DEFAULT FALSE,
    created_by INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Communication logs
CREATE TABLE communications (
    communication_id SERIAL PRIMARY KEY,
    patient_id INTEGER REFERENCES patients(patient_id),
    communication_type VARCHAR(50), -- Email, SMS, Call, Portal_Message
    direction VARCHAR(20), -- Incoming, Outgoing
    subject VARCHAR(255),
    content TEXT,
    sent_date TIMESTAMP,
    received_date TIMESTAMP,
    sent_by INTEGER,
    received_by INTEGER,
    status VARCHAR(20), -- Sent, Delivered, Read, Failed
    response_required BOOLEAN DEFAULT FALSE,
    response_status VARCHAR(20) -- Pending, Completed
);

```java
// Custom Metrics with Micrometer
@Component
public class HMSMetrics {
    
    private final Counter patientRegistrations;
    private final Timer appointmentBookingTime;
    private final Gauge activeUsers;
    private final Counter billingTransactions;
    
    public HMSMetrics(MeterRegistry meterRegistry) {
        this.patientRegistrations = Counter.builder("hms.patients.registered")
            .description("Number of patients registered")
            .tag("hospital", "default")
            .register(meterRegistry);
            
        this.appointmentBookingTime = Timer.builder("hms.appointments.booking.time")
            .description("Time taken to book appointments")
            .register(meterRegistry);
            
        this.activeUsers = Gauge.builder("hms.users.active")
            .description("Number of active users")
            .register(meterRegistry, this, HMSMetrics::getActiveUserCount);
            
        this.billingTransactions = Counter.builder("hms.billing.transactions")
            .description("Number of billing transactions")
            .register(meterRegistry);
    }
    
    public void recordPatientRegistration(String hospitalId) {
        patientRegistrations.increment(Tags.of("hospital", hospitalId));
    }
    
    public void recordAppointmentBooking(Duration duration) {
        appointmentBookingTime.record(duration);
    }
    
    private double getActiveUserCount() {
        // Implementation to count active sessions
        return sessionService.getActiveUserCount();
    }
}
```

```sql
SELECT 
    DATE_TRUNC('month', bill_date) as month,
    COUNT(*) as total_bills,
    SUM(total_amount) as total_revenue,
    SUM(paid_amount) as total_paid,
    AVG(total_amount) as avg_bill_amount
```

```javascript
// Secure Local Storage
class SecureStorage {
  static encrypt(data) {
    return btoa(JSON.stringify(data));
  }
  
  static decrypt(encryptedData) {
    try {
      return JSON.parse(atob(encryptedData));
    } catch {
      return null;
    }
  }
  
  static setItem(key, value) {
    const encrypted = this.encrypt(value);
    localStorage.setItem(key, encrypted);
  }
  
  static getItem(key) {
    const encrypted = localStorage.getItem(key);
    return encrypted ? this.decrypt(encrypted) : null;
  }
}
```

```java
// Custom Logging Aspect
@Aspect
@Component
@Slf4j
public class LoggingAspect {

    @Around("@annotation(Loggable)")
    public Object logExecutionTime(ProceedingJoinPoint joinPoint) throws Throwable {
        long start = System.currentTimeMillis();
        
        String className = joinPoint.getSignature().getDeclaringTypeName();
        String methodName = joinPoint.getSignature().getName();
        Object[] args = joinPoint.getArgs();
        
        try {
            log.info("Executing {}.{}() with args: {}", className, methodName, args);
            Object result = joinPoint.proceed();
            
            long executionTime = System.currentTimeMillis() - start;
            log.info("Completed {}.{}() in {} ms", className, methodName, executionTime);
            
            return result;
        } catch (Exception e) {
            long executionTime = System.currentTimeMillis() - start;
            log.error("Exception in {}.{}() after {} ms: {}", 
                className, methodName, executionTime, e.getMessage(), e);
            throw e;
        }
    }
    
    @Before("@annotation(AuditLog)")
    public void auditLog(JoinPoint joinPoint) {
        String action = joinPoint.getSignature().getName();
        String entityType = joinPoint.getSignature().getDeclaringTypeName();
        
        // Get current user and hospital context
        String userId = SecurityContextHolder.getContext().getAuthentication().getName();
        String hospitalId = TenantContext.getCurrentHospitalId();
        
        auditService.logAction(userId, hospitalId, action, entityType);
    }
}
```

```java
// Spring Boot Authentication Flow
@RestController
public class AuthController {
    
    @PostMapping("/api/auth/login")
    public ResponseEntity<AuthResponse> login(@RequestBody LoginRequest request) {
        // Validate credentials
        User user = userService.authenticate(request.getUsername(), request.getPassword());
        
        // Get user's hospital access
        List<UserHospitalAccess> hospitalAccess = userService.getHospitalAccess(user.getId());
        
        // Generate JWT token with hospital context
        String token = jwtService.generateToken(user, hospitalAccess);
        
        return ResponseEntity.ok(new AuthResponse(token, user, hospitalAccess));
    }
    
    @PostMapping("/api/auth/switch-hospital")
    public ResponseEntity<AuthResponse> switchHospital(@RequestBody SwitchHospitalRequest request) {
        // Validate hospital access
        userService.validateHospitalAccess(getCurrentUser().getId(), request.getHospitalId());
        
        // Update JWT token with new hospital context
        String newToken = jwtService.updateHospitalContext(getCurrentToken(), request.getHospitalId());
        
        return ResponseEntity.ok(new AuthResponse(newToken));
    }
}
```

```javascript
// contexts/AuthContext.js
import React, { createContext, useContext, useReducer, useEffect } from 'react';
import { authService } from '../services/auth.service';

const AuthContext = createContext();

const authReducer = (state, action) => {
  switch (action.type) {
    case 'LOGIN_SUCCESS':
      return {
        ...state,
        isAuthenticated: true,
        user: action.payload.user,
        hospitalAccess: action.payload.hospitalAccess,
        loading: false
      };
    case 'LOGOUT':
      return {
        ...state,
        isAuthenticated: false,
        user: null,
        hospitalAccess: [],
        loading: false
      };
    case 'SET_LOADING':
      return { ...state, loading: action.payload };
    default:
      return state;
  }
};

export const AuthProvider = ({ children }) => {
  const [state, dispatch] = useReducer(authReducer, {
    isAuthenticated: false,
    user: null,
    hospitalAccess: [],
    loading: true
  });

  useEffect(() => {
    const token = localStorage.getItem('token');
    if (token) {
      authService.validateToken(token)
        .then(response => {
          dispatch({
            type: 'LOGIN_SUCCESS',
            payload: response.data
          });
        })
        .catch(() => {
          localStorage.removeItem('token');
          dispatch({ type: 'SET_LOADING', payload: false });
        });
    } else {
      dispatch({ type: 'SET_LOADING', payload: false });
    }
  }, []);

  const login = async (credentials) => {
    try {
      const response = await authService.login(credentials);
      localStorage.setItem('token', response.data.token);
      dispatch({
        type: 'LOGIN_SUCCESS',
        payload: response.data
      });
      return response.data;
    } catch (error) {
      throw error;
    }
  };

  const logout = () => {
    localStorage.removeItem('token');
    localStorage.removeItem('currentHospitalId');
    dispatch({ type: 'LOGOUT' });
  };

  return (
    <AuthContext.Provider value={{ ...state, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within an AuthProvider');
  }
  return context;
};

// contexts/HospitalContext.js
import React, { createContext, useContext, useState, useEffect } from 'react';
import { hospitalService } from '../services/hospital.service';
import { useAuth } from './AuthContext';

const HospitalContext = createContext();

export const HospitalProvider = ({ children }) => {
  const [currentHospital, setCurrentHospital] = useState(null);
  const [hospitals, setHospitals] = useState([]);
  const { user, hospitalAccess } = useAuth();

  useEffect(() => {
    if (hospitalAccess?.length > 0) {
      setHospitals(hospitalAccess.map(access => access.hospital));
      
      // Set default hospital
      const savedHospitalId = localStorage.getItem('currentHospitalId');
      if (savedHospitalId) {
        const hospital = hospitalAccess.find(access => 
          access.hospital.id === parseInt(savedHospitalId)
        )?.hospital;
        if (hospital) {
          setCurrentHospital(hospital);
          return;
        }
      }
      
      // Set first hospital as default
      const defaultAccess = hospitalAccess.find(access => access.isDefaultHospital) 
        || hospitalAccess[0];
      setCurrentHospital(defaultAccess.hospital);
    }
  }, [hospitalAccess]);

  const switchHospital = async (hospitalId) => {
    try {
      await hospitalService.switchHospital(hospitalId);
      const hospital = hospitals.find(h => h.id === hospitalId);
      setCurrentHospital(hospital);
      localStorage.setItem('currentHospitalId', hospitalId.toString());
    } catch (error) {
      console.error('Failed to switch hospital:', error);
    }
  };

  return (
    <HospitalContext.Provider value={{
      currentHospital,
      hospitals,
      switchHospital
    }}>
      {children}
    </HospitalContext.Provider>
  );
};

export const useHospital = () => {
  const context = useContext(HospitalContext);
  if (!context) {
    throw new Error('useHospital must be used within a HospitalProvider');
  }
  return context;
};
```

## 🏥 Project Overview

Simple HMS is a comprehensive multi-tenant hospital management system designed to streamline healthcare operations across multiple facilities within a single organization. The system supports role-based access control and enables healthcare professionals to work seamlessly across different hospitals with unified login credentials.

### Key Capabilities
- **Multi-Hospital Management**: Single organization managing multiple clinics with schema-based data separation
- **Cross-Facility Practice**: Healthcare professionals can work at multiple hospitals with unified access
- **Role-Based Security**: Granular permissions for Admin, Doctor, Receptionist, and Medicine Clerk
- **Complete Patient Lifecycle**: From registration through treatment to follow-up care
- **Integrated Operations**: Appointments, billing, inventory, and reporting in one system

---

## 🛠 Tech Stack

- **Frontend**: React.js with Material-UI/Tailwind CSS
- **Backend**: Spring Boot (Java)
- **Database**: PostgreSQL with multi-tenant schema architecture
- **Authentication**: JWT with role-based access control
- **File Storage**: Local/Cloud storage for documents and images
- **Notifications**: Email/SMS integration (SendGrid/Twilio)

---

## 🏗 Database Architecture

### Multi-Tenant Design
- **Schema-per-tenant**: Each hospital gets its own PostgreSQL schema
- **Shared Schema**: Common organizational data and cross-hospital functionality
- **Dynamic Schema Creation**: Automated schema generation for new hospitals
